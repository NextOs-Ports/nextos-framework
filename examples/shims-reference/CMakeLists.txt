cmake_minimum_required(VERSION 3.16)
project(nextos_shim_reference VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(shim_reference STATIC shims.c)
target_include_directories(shim_reference PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(shim-reference-nextos main.c)
target_link_libraries(shim-reference-nextos PRIVATE shim_reference Threads::Threads)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(shim_reference PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(shim-reference-nextos PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
if(NOT CMAKE_CROSSCOMPILING)
  add_test(NAME explicit-shim-contracts COMMAND shim-reference-nextos)
endif()
